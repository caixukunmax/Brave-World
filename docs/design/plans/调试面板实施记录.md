namespace ClinetCSharp;

public static class RoleControlCenterXResolver
{
    public static float ResolveX(float manualX, bool centerX)
    {
        return centerX ? 0f : manualX;
    }

    public static bool IsManualXEditable(bool centerX)
    {
        return !centerX;
    }
}
